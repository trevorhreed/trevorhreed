Generate PIDs


const createPid = () => {
  const digits = 'abcdefghjkmnpqrstuvwxyz23456789';
  const uuid = crypto.randomUUID().replace(/-/g, '');
  let num = BigInt('0x' + uuid);
  let result = '';
  while (num > 0n) {
    result = digits[num % 31n] + result;
    num /= 31n;
  }
  return result.padStart(26, digits[0]);
}
